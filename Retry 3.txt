local function SafeCall(func)
    local success, err = pcall(func)
    if success then
        Cache.Stats.TotalFired = Cache.Stats.TotalFired + 1
    end
    return success
end


-- FULL GAME SCAN (every folder)
local function ScanAllRemotes()
    Log("Scanning EVERY remote in the ENTIRE game...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.RemoteFunctions = {}


    local locations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace
    }


    local function DeepScanFolder(folder)
        for _, obj in ipairs(folder:GetChildren()) do
            if obj:IsA("RemoteEvent") then
                table.insert(Cache.AllRemoteEvents, obj)
                if obj.Name == "TouchEvent" then
                    table.insert(Cache.TouchEvents, obj)
                end
            elseif obj:IsA("RemoteFunction") then
                table.insert(Cache.RemoteFunctions, obj)
            end
            DeepScanFolder(obj)
        end
    end


    for _, location in ipairs(locations) do
        pcall(function()
            for _, obj in ipairs(location:GetDescendants()) do
                -- Handled by DeepScan
            end
            DeepScanFolder(location)
        end)
    end


    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    Log(string.format("Found %d RemoteEvents (%d TouchEvents) and %d RemoteFunctions", 
        #Cache.AllRemoteEvents, #Cache.TouchEvents, #Cache.RemoteFunctions))
end


local function ScanTriggerParts()
    Log("Scanning for trigger parts...")
    Cache.TriggerParts = {}


    for _, obj in ipairs(Services.Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            if name:find("trip") or name:find("trigger") or name:find("touch") or 
               name:find("seat") or name:find("slide") or name:find("platform") or
               name:find("button") or name:find("lever") or name:find("zone") or
               name:find("teleport") or name:find("damage") or name:find("kill") or
               obj:FindFirstChildOfClass("TouchTransmitter") then
                table.insert(Cache.TriggerParts, obj)
            end
        end
    end
    Log(string.format("Found %d trigger parts", #Cache.TriggerParts))
end


local function GetOtherPlayers()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}


    for _, player in ipairs(Services.Players:GetPlayers()) do
        if player \~= LocalPlayer and player.Character then
            local char = player.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            
            if hrp and humanoid then
                table.insert(Cache.OtherPlayers, {
                    Player = player,
                    Character = char,
                    HRP = hrp,
                    Humanoid = humanoid,
                    Head = char:FindFirstChild("Head"),
                    Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                    Parts = {}
                })
                
                for _, part in ipairs(char:GetChildren()) do
                    if part:IsA("BasePart") then
                        table.insert(Cache.OtherPlayers[#Cache.OtherPlayers].Parts, part)
                    end
                end
            end
        end
    end


    local npcFolders = {
        Services.Workspace:FindFirstChild("Customers"),
        Services.Workspace:FindFirstChild("NPCs"),
        Services.Workspace:FindFirstChild("Enemies")
    }
    
    for _, folder in ipairs(npcFolders) do
        if folder then
            for _, npc in ipairs(folder:GetChildren()) do
                local hrp = npc:FindFirstChild("HumanoidRootPart")
                local humanoid = npc:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    table.insert(Cache.NPCs, {
                        Character = npc,
                        HRP = hrp,
                        Humanoid = humanoid
                    })
                end
            end
        end
    end


    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end


local function MultiArgFire(remote, target)
    if not remote then return 0 end
    local count = 0


    local targetsToFire = {}
    if target.HRP then table.insert(targetsToFire, target.HRP) end
    if target.Character then table.insert(targetsToFire, target.Character) end
    if target.Humanoid then table.insert(targetsToFire, target.Humanoid) end
    if target.Player then table.insert(targetsToFire, target.Player) end
    if target.Player then table.insert(targetsToFire, target.Player.UserId) end
    table.insert(targetsToFire, target.HRP)
    table.insert(targetsToFire, target.Character)
    table.insert(targetsToFire, "activate")
    table.insert(targetsToFire, true)
    table.insert(targetsToFire, {player = target.Player or target.Character})


    for i = 1, math.min(5, #targetsToFire) do
        SafeCall(function()
            remote:FireServer(targetsToFire[i])
            count = count + 1
        end)
    end
    return count
end


local function GetPriorityRemotes()
    local priorities = {}
    local priorityNames = {
        "Teleport", "Warp", "Reset", "Kick", "Ban", "Wipe", "Delete", "Kill", "Explode", "Boom", "Freeze", "Unfreeze"
    }


    for _, remote in ipairs(Cache.AllRemoteEvents) do
        local name = remote.Name:lower()
        for _, prio in ipairs(priorityNames) do
            if name:find(prio:lower()) then
                table.insert(priorities, 1, remote)
                break
            end
        end
    end
    return priorities
end


local function FireRemotesOnOthers()
    if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then return 0 end


    local fireCount = 0
    local allTargets = {}
    for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end


    local now = tick()


    local inAutoSetup = Config.AutoSetupPhase and (now - Cache.LastFireTime < 2)
    local rate = Config.RateLimitEnabled and 20 or 9999
    local targetCountPerBurst = math.clamp(rate * 0.6, 10, 25)


    local remainingInBurst = targetCountPerBurst - (now - Cache.LastFireTime) * rate
    local fireThisBurst = math.min(remainingInBurst, targetCountPerBurst)


    local toFire = {}
    if #Cache.AllRemoteEvents > 0 then
        local toFireCount = math.min(15, math.max(10, fireThisBurst))
        if Config.ChaosMode then
            toFire = GetPriorityRemotes()
        else
            for i = 1, toFireCount do
                table.insert(toFire, Cache.AllRemoteEvents[i % #Cache.AllRemoteEvents + 1])
            end
        end
    end


    if #toFire > 0 and Config.ChaosMode then
        toFire = GetPriorityRemotes()
    end


    for _, remote in ipairs(toFire) do
        for _, target in ipairs(allTargets) do
            fireCount = fireCount + MultiArgFire(remote, target)


            if table.find(Cache.TouchEvents, remote) then
                for _, triggerPart in ipairs(Cache.TriggerParts) do
                    SafeCall(function()
                        remote:FireServer(target.Character, triggerPart)
                        fireCount = fireCount + 1
                    end)
                end
            end
        end
    end


    if fireThisBurst > 0 then
        Cache.LastFireTime = now
    end


    Cache.Stats.TotalFired = Cache.Stats.TotalFired + fireCount
    return fireCount
end
